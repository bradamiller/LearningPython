from XRPLib.reflectance import Reflectance
from XRPLib.differential_drive import DifferentialDrive
from XRPLib.board import Board
import time
import random

reflectance = Reflectance.get_default_reflectance()
drivetrain = DifferentialDrive.get_default_differential_drive()
board = Board.get_default_board()


# ===== LINE SENSOR =====
# Reads the two reflectance sensors and answers questions
# about them. Starts here with the two raw readings; over this
# module it grows to answer "how far off the line am I?" and
# "am I on a line? on a cross?" Nothing in here drives.
# Everything that drives asks these functions instead of the
# hardware.
THRESHOLD = 0.5          # use YOUR value from Lesson 2-1

def get_left():
    return reflectance.get_left()

def get_right():
    return reflectance.get_right()

def get_error():
    return get_left() - get_right()

def is_either_on_line():
    return get_left() > THRESHOLD or get_right() > THRESHOLD

def is_at_cross():
    return get_left() > THRESHOLD and get_right() > THRESHOLD

def set_threshold(new_value):
    global THRESHOLD
    THRESHOLD = new_value

def test_sensors():
    for i in range(50):
        print("Left:", round(get_left(), 2), " Right:", round(get_right(), 2),
              " Error:", round(get_error(), 2),
              " On line:", is_either_on_line(), " Cross:", is_at_cross())
        time.sleep(0.1)

def test_drive_to_line():
    while get_left() < THRESHOLD and get_right() < THRESHOLD:
        drivetrain.set_effort(0.3, 0.3)
    drivetrain.stop()
    print("Line detected!")

def test_bounce():
    bounce_count = 0
    while True:
        if is_either_on_line():
            drivetrain.stop()
            bounce_count = bounce_count + 1
            print("Bounce", bounce_count)
            drivetrain.turn(180)
        else:
            drivetrain.set_effort(0.3, 0.3)

def test_random_bounce():
    speed = 0.3
    while True:
        if is_either_on_line():
            drivetrain.stop()
            angle = random.randint(90, 270)
            speed = random.randint(20, 50) / 100
            print("Turning", angle, "degrees, then speed", speed)
            drivetrain.turn(angle)
        else:
            drivetrain.set_effort(speed, speed)


# ===== LINE TRACKING =====
# Drives the robot along the line using the LINE SENSOR
# functions above. Starts with the proportional follower: read
# the error, scale it by KP, steer with arcade(). Grows over
# the module into following to a cross, parking on it, and
# turning to the next line.
BASE_EFFORT = 0.4        # forward speed while following
KP = 0.5                 # proportional gain -- tune this!

def test_follow():
    while True:
        error = get_error()
        correction = error * KP
        drivetrain.arcade(BASE_EFFORT, correction)
        time.sleep(0.01)

def test_sweep():
    for i in range(500):
        error = get_error()
        correction = error * KP
        if i % 50 == 0:
            print(f"i={i}  error={error:.2f}  correction={correction:.2f}")
        time.sleep(0.02)

# --- Lesson 2-8: follow to a cross ---
# track_to_intersection() is the follower from test_follow()
# with a reason to stop: it follows until is_at_cross() says
# the robot is on a crossing line. Lesson 2-9 adds the
# functions that park on the cross and turn to the next line.

def track_to_intersection():
    while not is_at_cross():
        correction = get_error() * KP
        drivetrain.arcade(BASE_EFFORT, correction)
    drivetrain.stop()

def test_cross_moving():
    while not is_at_cross():
        drivetrain.set_effort(0.3, 0.3)
    drivetrain.stop()
    print("Cross!")

def test_track():
    track_to_intersection()
    print("Done!")

def test_reverse():
    track_to_intersection()
    drivetrain.turn(180)
    track_to_intersection()
    print("Done!")

def test_count():
    for count in range(4):
        track_to_intersection()
        print("Cross #", count + 1)
        drivetrain.turn(180)
    print("Done!")

# --- Lesson 2-9: park on the cross, turn to the next line ---
# Completes the driving toolkit begun in Lesson 2-5.
# track_to_intersection() stops with the SENSORS on the cross;
# drive_past_intersection() moves the WHEELS onto it. The two
# turns spin until a LINE SENSOR function sees the next line,
# so no turn depends on a timer. With these four, every
# program from here to Module 5 is a list of calls to this
# section.
TURN_EFFORT = 0.25       # spin speed during a turn

def drive_past_intersection():
    drivetrain.straight(8.75, 0.5)

def turn_clockwise():
    drivetrain.turn(-60, TURN_EFFORT)
    drivetrain.arcade(0, -TURN_EFFORT)
    while not is_either_on_line():
        pass
    drivetrain.stop()

def turn_counterclockwise():
    drivetrain.turn(60, TURN_EFFORT)
    drivetrain.arcade(0, TURN_EFFORT)
    while not is_either_on_line():
        pass
    drivetrain.stop()

def turn_around():
    # on the circle the cross is too short to stop the spin,
    # so one turn ends facing back the way it came
    turn_clockwise()

def test_park():
    track_to_intersection()
    drive_past_intersection()

def test_turn():
    track_to_intersection()
    drive_past_intersection()
    turn_clockwise()

def test_course():
    track_to_intersection()
    drive_past_intersection()
    turn_clockwise()
    track_to_intersection()
    print("Done!")

def test_turn_around():
    track_to_intersection()
    drive_past_intersection()
    turn_around()
    track_to_intersection()
    print("Done!")

# --- Lesson 2-10: final project ---
# The module project: lap the circle, reversing at the cross,
# four times, then stop. Nothing new is defined for it. It is
# a list of calls to the four driving functions above, which
# is the point: the sections did the work, the project
# assembles them.

def test_reversal():
    track_to_intersection()
    drive_past_intersection()
    turn_clockwise()      # one sensor-ended turn = 180 deg here
    track_to_intersection()
    print("Done!")

def test_final_project():
    print("Module 2 Final Project -- starting!")
    for leg in range(4):
        # follow the line to the cross
        print("Leg", leg + 1, "-- following to the cross...")
        track_to_intersection()
        # park the wheels on the cross
        drive_past_intersection()
        # turn around: one sensor-ended turn finds the circle behind
        print("Cross detected! Reversing...")
        turn_clockwise()
    print("Done! 4 reversals complete.")


# ===== GRID DRIVING =====
# Drives the taped grid one intersection at a time, using only
# the LINE TRACKING functions: a grid intersection is the
# cross from Module 2. To ARRIVE at an intersection is
# track_to_intersection() then drive_past_intersection(); a
# turn happens parked on one. Nothing here touches a sensor.

def test_first_intersection():
    print("Driving to the first intersection...")
    track_to_intersection()
    drive_past_intersection()
    print("Intersection reached!")

def test_arrive_and_turn():
    track_to_intersection()
    drive_past_intersection()
    print("Turning right...")
    turn_clockwise()
    print("Now facing a new direction!")

# --- Lesson 3-2: a run of intersections ---
# drive_intersections(count) is the arrival pair in a loop.
# It ends parked on the count-th intersection, so Lessons 3-3
# and 3-4 build every path out of it and the two turns.

def drive_intersections(count):
    """Arrive at count intersections in a row, printing
    which one it is each time. Ends parked on the last one
    with the sensors past it, ready to turn or to go on."""
    for i in range(count):
        print("  Intersection", i + 1, "of", count)
        track_to_intersection()
        drive_past_intersection()

def test_two_by_hand():
    track_to_intersection()
    drive_past_intersection()
    track_to_intersection()
    drive_past_intersection()
    print("Parked on the second intersection")

def test_drive_run():
    drive_intersections(3)
    print("Done! Parked on intersection 3")

# --- Lesson 3-3: turning on the grid ---
# No new function: a path is drive_intersections(), a turn,
# drive_intersections() again. The turn works because the
# helper leaves the wheels parked on the intersection.

def test_l_shape():
    drive_intersections(2)      # leg 1
    turn_clockwise()            # corner: right
    drive_intersections(2)      # leg 2
    print("L-shape complete!")

def test_my_path():
    drive_intersections(3)      # leg 1
    turn_counterclockwise()     # corner: left
    drive_intersections(1)      # leg 2
    print("Path complete!")

# --- Lesson 3-4: final project, the square ---
# The module project: four sides of SIDE intersections with a
# right turn at every corner, back to the start. Built from
# drive_intersections() and one turn; nothing new is defined.
SIDE = 2                 # intersections per side

def test_one_side():
    drive_intersections(SIDE)
    turn_clockwise()
    print("One side done")

def test_square():
    print("Module 3 Final Project -- Square Pattern!")
    for side in range(4):
        print("Side", side + 1, "of 4")
        drive_intersections(SIDE)
        print("  Turning right...")
        turn_clockwise()
    print("Square complete! Back at start.")

# ===== RUN =====
board.wait_for_button()

from XRPLib.reflectance import Reflectance
from XRPLib.differential_drive import DifferentialDrive
from XRPLib.board import Board
import time
import random

reflectance = Reflectance.get_default_reflectance()
drivetrain = DifferentialDrive.get_default_differential_drive()
board = Board.get_default_board()


# ===== LINE SENSOR (Lesson 2-7, class version) =====
# One LineSensor object owns the reflectance sensor and its
# threshold, and answers questions about the line: how far off
# it we are (get_error) and whether we are on a cross or on a
# line at all. Nothing in here drives. Everything that drives,
# from the next lesson on, asks this object.

class LineSensor:
    def __init__(self):
        self.reflectance = Reflectance.get_default_reflectance()
        self.threshold = 0.5

    def get_left(self):
        return self.reflectance.get_left()

    def get_right(self):
        return self.reflectance.get_right()

    def get_error(self):
        return self.get_left() - self.get_right()

    def is_at_cross(self):
        return self.get_left() > self.threshold and self.get_right() > self.threshold

    def is_either_on_line(self):
        return self.get_left() > self.threshold or self.get_right() > self.threshold

sensor = LineSensor()

def test_sensors():
    for i in range(50):
        print("Left:", round(sensor.get_left(), 2), " Right:", round(sensor.get_right(), 2),
              " Error:", round(sensor.get_error(), 2),
              " On line:", sensor.is_either_on_line(), " Cross:", sensor.is_at_cross())
        time.sleep(0.1)

def test_drive_to_line():
    while sensor.get_left() < sensor.threshold and sensor.get_right() < sensor.threshold:
        drivetrain.set_effort(0.3, 0.3)
    drivetrain.stop()
    print("Line detected!")

# ===== LINE TRACKING (class version) =====
# One LineTrack object drives the robot along the line. It HAS
# A LineSensor (built in __init__, so you never make one
# separately) and asks it the questions; it HAS the drivetrain
# and gives it the orders. Same four methods as the function
# version, same names, with tracker. in front.

class LineTrack:
    def __init__(self):
        self.sensor = LineSensor()           # has-a LineSensor
        self.drivetrain = DifferentialDrive.get_default_differential_drive()
        self.base_effort = 0.4
        self.kp = 0.5
        self.turn_effort = 0.25

    def track_to_intersection(self):
        while not self.sensor.is_at_cross():
            error = self.sensor.get_error()
            correction = error * self.kp
            self.drivetrain.arcade(self.base_effort, correction)
        self.drivetrain.stop()

    def drive_past_intersection(self):
        self.drivetrain.straight(8.75, 0.5)

    def turn_clockwise(self):
        self.drivetrain.turn(-60, self.turn_effort)
        self.drivetrain.arcade(0, -self.turn_effort)
        while not self.sensor.is_either_on_line():
            pass
        self.drivetrain.stop()

    def turn_counterclockwise(self):
        self.drivetrain.turn(60, self.turn_effort)
        self.drivetrain.arcade(0, self.turn_effort)
        while not self.sensor.is_either_on_line():
            pass
        self.drivetrain.stop()
