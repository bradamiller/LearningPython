# ===== MANHATTAN PLANNER =====
# Plans a route across the grid: given where the robot is and
# where it should go, returns the list of (x, y) tuples to
# drive to, across first and then up or down. Pure arithmetic --
# nothing in here touches a sensor or a motor, so every test
# in this section runs on screen with the robot sitting still.

def compute_manhattan_path(position, destination):
    """Compute a Manhattan path from position to destination.

    position:    an (x, y) tuple -- where the robot is now
    destination: an (x, y) tuple -- where it needs to go

    Returns a list of (x, y) tuples the robot should move to,
    not including position itself.
    """
    path = []
    x, y = position
    dest_x, dest_y = destination

    while x < dest_x:
        x = x + 1
        path.append((x, y))

    while x > dest_x:
        x = x - 1
        path.append((x, y))

    while y < dest_y:
        y = y + 1
        path.append((x, y))

    while y > dest_y:
        y = y - 1
        path.append((x, y))

    return path

HEADING_NAMES = ["N", "E", "S", "W"]    # HEADING_NAMES[2] -> "S"

def desired_heading(position, next_pos):
    """Return the heading (0-3) that points from position to
    next_pos, which must be one step away; raise if it is not."""
    x_diff = next_pos[0] - position[0]
    y_diff = next_pos[1] - position[1]
    if x_diff == 0 and y_diff ==  1: return 0   # North
    if y_diff == 0 and x_diff ==  1: return 1   # East
    if x_diff == 0 and y_diff == -1: return 2   # South
    if y_diff == 0 and x_diff == -1: return 3   # West
    raise ValueError(str(position) + " and " + str(next_pos) + " are not neighbors")

# --- Lesson 4-8: turn to a heading, drive a path ---
# turn_to() turns a quarter turn at a time, with the LINE
# TRACKING turns, until the robot faces the heading it
# needs. drive_path() is the loop: for each
# intersection, face it, arrive with the track-then-drive-past
# pair, and record the new position. Both hand back what they
# changed, because the caller keeps the only record.

def turn_to(heading, desired):
    """Turn clockwise a quarter turn at a time until heading
    is desired; return the new heading."""
    while heading != desired:
        turn_clockwise()
        heading = (heading + 1) % 4
    return heading

def drive_path(path, position, heading):
    """Turn and drive through every intersection in path, from
    position facing heading; return the final (position, heading)."""
    for next_pos in path:
        needed = desired_heading(position, next_pos)
        heading = turn_to(heading, needed)
        track_to_intersection()
        drive_past_intersection()
        position = next_pos
    return position, heading

# ===== GRID GRAPH =====
# The grid as a graph, in a dictionary: every intersection is
# a key, and its value is the list of neighbors it is joined
# to. The BFS PLANNER searches this graph. Starts with the
# open grid; Lesson 5-4 adds the blocked list, so a blocked
# intersection is neither a key nor anyone's neighbor.
ROWS = 4                 # the grid is ROWS x COLS intersections
COLS = 4

def build_graph(rows, cols, blocked):
    """Return the grid as a dictionary, leaving out every
    intersection in blocked: not a key, not anyone's neighbor."""
    graph = {}
    for x in range(cols):
        for y in range(rows):
            if (x, y) in blocked:
                continue
            neighbors = []
            for dx, dy in [(-1, 0), (1, 0), (0, -1), (0, 1)]:
                nx, ny = x + dx, y + dy
                if 0 <= nx < cols and 0 <= ny < rows and (nx, ny) not in blocked:
                    neighbors.append((nx, ny))
            graph[(x, y)] = neighbors
    return graph

# ===== BFS PLANNER =====
# The second planner. compute_bfs_path(position, destination)
# has the same shape as the MANHATTAN PLANNER's function and
# keeps the same two promises (told the position; the path
# leaves the start out), but it searches the GRID GRAPH ring
# by ring, so it routes AROUND whatever is in BLOCKED. Pure
# dictionary work: every test here runs with the robot still.

def compute_bfs_path(position, destination):
    """Shortest path from position to destination around
    BLOCKED, as the list of intersections to drive to."""
    graph = build_graph(ROWS, COLS, BLOCKED)
    if position not in graph or destination not in graph:
        return []
    previous = {position: None}
    to_visit = [position]

    while len(to_visit) > 0:
        current = to_visit.pop(0)
        if current == destination:
            break
        for neighbor in graph[current]:
            if neighbor not in previous:
                previous[neighbor] = current
                to_visit.append(neighbor)

    if destination not in previous:
        return []
    path = []
    current = destination
    while current != position:
        path.append(current)
        current = previous[current]
    path.reverse()
    return path

def bfs_rings(position):
    """Ring number of every intersection reachable from
    position, as a dictionary -- the search with no break."""
    graph = build_graph(ROWS, COLS, BLOCKED)
    if position not in graph:
        return {}
    rings = {position: 0}
    to_visit = [position]
    while len(to_visit) > 0:
        current = to_visit.pop(0)
        for neighbor in graph[current]:
            if neighbor not in rings:
                rings[neighbor] = rings[current] + 1
                to_visit.append(neighbor)
    return rings

def print_grid(rows, cols, blocked, numbers):
    """Draw the grid the way it lies in front of you: the top
    line is the far row, y = 0 is the bottom line. Each
    intersection shows its entry in numbers, X if it is
    blocked, . if it has no entry."""
    for i in range(rows):
        y = rows - 1 - i
        line = " " + str(y) + " | "
        for x in range(cols):
            if (x, y) in blocked:
                cell = "X"
            elif (x, y) in numbers:
                cell = numbers[(x, y)]
            else:
                cell = "."
            line = line + "{:>3}".format(cell)
        print(line)
    print("    " + "-" * (3 * cols + 1))
    footer = "     "
    for x in range(cols):
        footer = footer + "{:>3}".format(x)
    print(footer)

# Tests any planner by what must be true of its path, not by

def manhattan_distance(a, b):
    """Steps between a and b with nothing in the way."""
    return abs(a[0] - b[0]) + abs(a[1] - b[1])

def is_neighbor(a, b):
    """True when a and b are one step apart."""
    return manhattan_distance(a, b) == 1

def check_path(path, position, destination):
    """Return a list of everything wrong with path as a route
    from position to destination; [] means it is good."""
    problems = []
    if len(path) == 0:
        if position != destination:
            problems.append("empty path, but the robot is not at the destination")
        return problems
    if not is_neighbor(position, path[0]):
        problems.append("first step " + str(path[0]) + " is not next to the start " + str(position))
    if path[-1] != destination:
        problems.append("last step " + str(path[-1]) + " is not the destination " + str(destination))
    for i in range(1, len(path)):
        if not is_neighbor(path[i - 1], path[i]):
            problems.append("step " + str(path[i]) + " is not next to " + str(path[i - 1]))
    for step in path:
        if step in BLOCKED:
            problems.append("drives through blocked intersection " + str(step))
    return problems

def run_path_test(test_name, planner, position, destination, expected_steps):
    """Ask planner for a path, check the five properties, print
    PASS or FAIL with the path; return True on a pass."""
    path = planner(position, destination)
    problems = check_path(path, position, destination)
    if len(path) != expected_steps:
        problems.append("expected " + str(expected_steps) + " steps, got " + str(len(path)))
    if len(problems) == 0:
        print("  PASS ", test_name, "", path)
        return True
    print("  FAIL ", test_name, "", path)
    for problem in problems:
        print("          -", problem)
    return False

def run_no_path_test(test_name, planner, position, destination):

        return True
    print("  FAIL ", test_name, "", path)
    print("          - there is no way through, so the path should be empty")
    return False

# navigate_with_detection() is the 5-7 loop inside a while:
# plan, follow, and if something got in the way, plan again
# from here, until a whole path is driven with no reroute. It

def navigate_with_detection(position, heading, destination, rangefinder, obstacle_cm):
    """Drive from position (facing heading) to destination,
    re-planning around every obstacle the rangefinder finds;
    return (position, heading, step_count, reroute_count)."""
    step_count = 0
    reroute_count = 0

    arrived = False
    while not arrived:
        path = compute_bfs_path(position, destination)
        print("  Path from", position, ":", path)

        if len(path) == 0:
            print("No path to", destination, "!")
            return position, heading, step_count, reroute_count

        rerouted = False
        for next_pos in path:
            needed = desired_heading(position, next_pos)
            heading = turn_to(heading, needed)

            distance = rangefinder.distance()          # look before driving
            if distance < obstacle_cm:
                add_blocked(next_pos)
                print("  OBSTACLE at", next_pos, "! Rerouting...")
                reroute_count = reroute_count + 1
                rerouted = True
                break
            else:
                track_to_intersection()
                drive_past_intersection()
                position = next_pos
                step_count = step_count + 1

        if not rerouted:
            arrived = True

    return position, heading, step_count, reroute_count

# ===== WEIGHTED GRAPH =====
# The grid with a cost on every edge. A normal block costs 1;
# a slow segment costs what you measured in Lesson 6-1. The
# slow list is kept the way GRID GRAPH keeps BLOCKED, and
# weight() is the one question every planner asks of it.
# path_cost() adds a route up; check_cost() is a sixth
# property for the BFS PLANNER's harness, and BFS fails it.
SLOW_EDGES = []          # (a, b, cost), one per slow segment

def add_slow_edge(a, b, cost):
    """Record the segment between a and b as costing cost."""
    SLOW_EDGES.append((a, b, cost))

def clear_slow_edges():
    """Empty SLOW_EDGES in place, as clear_blocked() does."""
    while len(SLOW_EDGES) > 0:
        SLOW_EDGES.pop()

def weight(a, b):
    """Cost of driving from a to b: the slow cost if that
    segment is in SLOW_EDGES, in either order; otherwise 1."""
    for (end1, end2, cost) in SLOW_EDGES:
        if (a == end1 and b == end2) or (a == end2 and b == end1):
            return cost
    return 1

def path_cost(position, path):
    """Total cost of driving path from position: the weight
    of every segment, starting with position to path[0]."""
    total = 0
    here = position
    for step in path:
        total = total + weight(here, step)
        here = step
    return total

def check_cost(test_name, position, destination, path, expected_cost):
    """Print PASS if path costs expected_cost, else FAIL with
    the cost and the path; return True on a pass."""
    cost = path_cost(position, path)
    if cost == expected_cost:
        print("  PASS ", test_name, "", path, " cost:", cost)
        return True
    print("  FAIL ", test_name, "", path, " cost:", cost)
    print("          - the cheapest route costs", expected_cost)
    return False

# ===== DIJKSTRA PLANNER =====
# The third planner, made by copying the BFS PLANNER's
# function and changing two things in the middle: take the
# CLOSEST waiting intersection, not the oldest, and add the
# WEIGHTED GRAPH's weight() instead of counting a step,
# keeping the cheaper total when a place is reached twice.
# Same shape, same promises; the 5-6 harness tests all three.

def closest(intersections, distance):
    """Return the entry of intersections with the smallest
    value in distance: a plain find-the-minimum loop."""
    best = intersections[0]
    for node in intersections:
        if distance[node] < distance[best]:
            best = node
    return best

def compute_dijkstra_path(position, destination):
    """Cheapest path from position to destination around
    BLOCKED, by weight(), as the intersections to drive to."""
    graph = build_graph(ROWS, COLS, BLOCKED)
    if position not in graph or destination not in graph:
        return []
    previous = {position: None}
    distance = {position: 0}
    to_visit = [position]

    while len(to_visit) > 0:
        current = closest(to_visit, distance)
        to_visit.remove(current)
        if current == destination:
            break
        for neighbor in graph[current]:
            new_distance = distance[current] + weight(current, neighbor)
            if neighbor not in distance or new_distance < distance[neighbor]:
                distance[neighbor] = new_distance
                previous[neighbor] = current
                if neighbor not in to_visit:
                    to_visit.append(neighbor)

    if destination not in previous:
        return []
    path = []
    current = destination
    while current != position:
        path.append(current)
        current = previous[current]
    path.reverse()
    return path

def dijkstra_distances(position):
    """Cheapest cost to every reachable intersection, as a
    dictionary -- the search with no break, like bfs_rings()."""
    graph = build_graph(ROWS, COLS, BLOCKED)
    if position not in graph:
        return {}
    distance = {position: 0}
    to_visit = [position]
    while len(to_visit) > 0:
        current = closest(to_visit, distance)
        to_visit.remove(current)
        for neighbor in graph[current]:
            new_distance = distance[current] + weight(current, neighbor)
            if neighbor not in distance or new_distance < distance[neighbor]:
                distance[neighbor] = new_distance
                if neighbor not in to_visit:
                    to_visit.append(neighbor)
    return distance

def compare(position, destination):
    """Print BFS's and Dijkstra's answers for one trip:
    steps, cost and path, one line each."""
    for (name, planner) in [("BFS", compute_bfs_path),
                            ("Dijkstra", compute_dijkstra_path)]:
        path = planner(position, destination)
        print(" ", "{:<9}".format(name), "steps:", len(path),
              " cost:", path_cost(position, path), " ", path)

# ===== MANHATTAN PLANNER (class version) =====
# One Manhattan object plans routes on the grid, across
# first and then up or down. It holds NO state: compute_path()
# is told the robot's position every call. Pure arithmetic.

class Manhattan:
    """The Lesson 4-5 planner as a class.

    It holds NO state: compute_path() is told the robot's position
    every time. Manhattan is a class only so that it has the same
    shape as the Module 5 and 6 planners (BFS, Dijkstra) -- then one
    main program and one test harness work with any of them by
    changing one name.
    """

    def __init__(self):
        pass        # nothing to set up

    def compute_path(self, position, destination):
        path = []
        x, y = position
        dest_x, dest_y = destination

        while x < dest_x:
            x = x + 1
            path.append((x, y))

        while x > dest_x:
            x = x - 1
            path.append((x, y))

        while y < dest_y:
            y = y + 1
            path.append((x, y))

        while y > dest_y:
            y = y - 1
            path.append((x, y))

        return path

    def add_blocked(self, intersection):
        # Manhattan cannot route around obstacles. This exists
        # so every planner can be tested the same way. It
        # deliberately does nothing.
        pass

    def clear_blocked(self):
        pass

    def add_slow_edge(self, a, b, cost):
        pass        # does not plan by cost; accepts the call

    def clear_slow_edges(self):
        pass

# ===== NAVIGATOR (class version) =====
# One Navigator object owns the robot's position and heading.
# It HAS a LineTrack (built in __init__) and gives it the
# driving orders; it is told where to go and keeps its own
# record of where it got to. Same three methods as the
# function version, same names, with navigator. in front.
HEADING_NAMES = ["N", "E", "S", "W"]    # HEADING_NAMES[2] -> "S"

class Navigator:
    def __init__(self, start, heading):
        self.position = start
        self.heading = heading
        self.line_track = LineTrack()        # has-a LineTrack

    def desired_heading(self, next_pos):
        x_diff = next_pos[0] - self.position[0]
        y_diff = next_pos[1] - self.position[1]
        if x_diff == 0 and y_diff ==  1: return 0   # North
        if y_diff == 0 and x_diff ==  1: return 1   # East
        if x_diff == 0 and y_diff == -1: return 2   # South
        if y_diff == 0 and x_diff == -1: return 3   # West
        raise ValueError(str(self.position) + " and " + str(next_pos) + " are not neighbors")

    def turn_to(self, desired):
        while self.heading != desired:
            self.line_track.turn_clockwise()
            self.heading = (self.heading + 1) % 4

    def drive_path(self, path):
        for next_pos in path:
            needed = self.desired_heading(next_pos)
            self.turn_to(needed)
            self.line_track.track_to_intersection()
            self.line_track.drive_past_intersection()
            self.position = next_pos

    def navigate_with_detection(self, destination, rangefinder, obstacle_cm):
        """Drive from self.position (facing self.heading) to
        destination, re-planning around every obstacle the
        rangefinder finds; return (step_count, reroute_count)."""
        step_count = 0
        reroute_count = 0

        arrived = False
        while not arrived:
            path = planner.compute_path(self.position, destination)
            print("  Path from", self.position, ":", path)

            if len(path) == 0:
                print("No path to", destination, "!")
                return step_count, reroute_count

            rerouted = False
            for next_pos in path:
                needed = self.desired_heading(next_pos)
                self.turn_to(needed)

                distance = rangefinder.distance()          # look before driving
                if distance < obstacle_cm:
                    planner.add_blocked(next_pos)
                    print("  OBSTACLE at", next_pos, "! Rerouting...")
                    reroute_count = reroute_count + 1
                    rerouted = True
                    break
                else:
                    self.line_track.track_to_intersection()
                    self.line_track.drive_past_intersection()
                    self.position = next_pos
                    step_count = step_count + 1

            if not rerouted:
                arrived = True

        return step_count, reroute_count

# ===== GRID GRAPH =====
# The grid as a graph, in a dictionary: every intersection is
# a key, and its value is the list of neighbors it is joined
# to. The BFS PLANNER searches this graph. Starts with the
# open grid; Lesson 5-4 adds the blocked list, so a blocked
# intersection is neither a key nor anyone's neighbor.
ROWS = 4                 # the grid is ROWS x COLS intersections
COLS = 4

def build_graph(rows, cols, blocked):
    """Return the grid as a dictionary, leaving out every
    intersection in blocked: not a key, not anyone's neighbor."""
    graph = {}
    for x in range(cols):
        for y in range(rows):
            if (x, y) in blocked:
                continue
            neighbors = []
            for dx, dy in [(-1, 0), (1, 0), (0, -1), (0, 1)]:
                nx, ny = x + dx, y + dy
                if 0 <= nx < cols and 0 <= ny < rows and (nx, ny) not in blocked:
                    neighbors.append((nx, ny))
            graph[(x, y)] = neighbors
    return graph

# ===== BFS PLANNER (class version) =====
# One BFS object holds the grid size and ITS OWN blocked list
# and plans around it: compute_path() builds the graph from
# self.blocked on every call. It holds NO position (Rule 1).
# Same names as the function version, with planner. in front.

class BFS:
    def __init__(self, rows=4, cols=4):
        self.rows = rows
        self.cols = cols
        self.blocked = []

    def add_blocked(self, intersection):
        if intersection not in self.blocked:
            self.blocked.append(intersection)

    def clear_blocked(self):
        self.blocked = []

    def compute_path(self, position, destination):
        graph = build_graph(self.rows, self.cols, self.blocked)
        if position not in graph or destination not in graph:
            return []
        previous = {position: None}
        to_visit = [position]

        while len(to_visit) > 0:
            current = to_visit.pop(0)
            if current == destination:
                break
            for neighbor in graph[current]:
                if neighbor not in previous:
                    previous[neighbor] = current
                    to_visit.append(neighbor)

        if destination not in previous:
            return []
        path = []
        current = destination
        while current != position:
            path.append(current)
            current = previous[current]
        path.reverse()
        return path

    def add_slow_edge(self, a, b, cost):
        pass        # does not plan by cost; accepts the call

    def clear_slow_edges(self):
        pass

def bfs_rings(position):
    """Ring number of every intersection reachable from
    position, as a dictionary -- the search with no break."""
    graph = build_graph(ROWS, COLS, planner.blocked)
    if position not in graph:
        return {}
    rings = {position: 0}
    to_visit = [position]
    while len(to_visit) > 0:
        current = to_visit.pop(0)
        for neighbor in graph[current]:
            if neighbor not in rings:
                rings[neighbor] = rings[current] + 1
                to_visit.append(neighbor)
    return rings

def print_grid(rows, cols, blocked, numbers):
    """Draw the grid the way it lies in front of you: the top
    line is the far row, y = 0 is the bottom line. Each
    intersection shows its entry in numbers, X if it is
    blocked, . if it has no entry."""
    for i in range(rows):
        y = rows - 1 - i
        line = " " + str(y) + " | "
        for x in range(cols):
            if (x, y) in blocked:
                cell = "X"
            elif (x, y) in numbers:
                cell = numbers[(x, y)]
            else:
                cell = "."
            line = line + "{:>3}".format(cell)
        print(line)
    print("    " + "-" * (3 * cols + 1))
    footer = "     "
    for x in range(cols):
        footer = footer + "{:>3}".format(x)
    print(footer)

# --- Lesson 5-6: the property harness ---
# Tests any planner by what must be true of its path, not by
# one expected list, because BFS has many right answers.
# run_all_tests() is handed a planner FUNCTION and runs the
# six cases through it: one program tests Manhattan and BFS.
# Module 4's run_test() stays; exact match is right there.
HARNESS_BLOCKED = []

def block(planner, intersection):
    """Tell planner about intersection AND note it in
    HARNESS_BLOCKED, so check_path() can see it."""
    planner.add_blocked(intersection)
    if intersection not in HARNESS_BLOCKED:
        HARNESS_BLOCKED.append(intersection)

def clear(planner):
    """Tell planner to forget its blocks and slow edges,
    and forget ours."""
    planner.clear_blocked()
    planner.clear_slow_edges()
    while len(HARNESS_BLOCKED) > 0:
        HARNESS_BLOCKED.pop()
    clear_slow_edges()

def slow(planner, a, b, cost):
    """Tell planner about the slow edge AND note it in
    SLOW_EDGES, so path_cost() can see it."""
    planner.add_slow_edge(a, b, cost)
    add_slow_edge(a, b, cost)

def manhattan_distance(a, b):
    """Steps between a and b with nothing in the way."""
    return abs(a[0] - b[0]) + abs(a[1] - b[1])

def is_neighbor(a, b):
    """True when a and b are one step apart."""
    return manhattan_distance(a, b) == 1

def check_path(path, position, destination):
    """Return a list of everything wrong with path as a route
    from position to destination; [] means it is good."""
    problems = []
    if len(path) == 0:
        if position != destination:
            problems.append("empty path, but the robot is not at the destination")
        return problems
    if not is_neighbor(position, path[0]):
        problems.append("first step " + str(path[0]) + " is not next to the start " + str(position))
    if path[-1] != destination:
        problems.append("last step " + str(path[-1]) + " is not the destination " + str(destination))
    for i in range(1, len(path)):
        if not is_neighbor(path[i - 1], path[i]):
            problems.append("step " + str(path[i]) + " is not next to " + str(path[i - 1]))
    for step in path:
        if step in HARNESS_BLOCKED:
            problems.append("drives through blocked intersection " + str(step))
    return problems

def run_path_test(test_name, planner, position, destination, expected_steps):
    """Ask planner for a path, check the five properties, print
    PASS or FAIL with the path; return True on a pass."""
    path = planner.compute_path(position, destination)
    problems = check_path(path, position, destination)
    if len(path) != expected_steps:
        problems.append("expected " + str(expected_steps) + " steps, got " + str(len(path)))
    if len(problems) == 0:
        print("  PASS ", test_name, "", path)
        return True
    print("  FAIL ", test_name, "", path)
    for problem in problems:
        print("          -", problem)
    return False

def run_no_path_test(test_name, planner, position, destination):
    """For a walled-off destination the only right answer is
    []; print PASS or FAIL and return True on a pass."""
    path = planner.compute_path(position, destination)
    if len(path) == 0:
        print("  PASS ", test_name, "", path)
        return True
    print("  FAIL ", test_name, "", path)
    print("          - there is no way through, so the path should be empty")
    return False

# The robot fills BLOCKED itself. Before each step of a path

# ===== WEIGHTED GRAPH =====
# The grid with a cost on every edge. A normal block costs 1;
# a slow segment costs what you measured in Lesson 6-1. The
# slow list is kept the way GRID GRAPH keeps BLOCKED, and
# weight() is the one question every planner asks of it.
# path_cost() adds a route up; check_cost() is a sixth
# property for the BFS PLANNER's harness, and BFS fails it.
SLOW_EDGES = []          # (a, b, cost), one per slow segment

def add_slow_edge(a, b, cost):
    """Record the segment between a and b as costing cost."""
    SLOW_EDGES.append((a, b, cost))

def clear_slow_edges():
    """Empty SLOW_EDGES in place, as planner.clear_blocked() does."""
    while len(SLOW_EDGES) > 0:
        SLOW_EDGES.pop()

def weight(a, b):
    """Cost of driving from a to b: the slow cost if that
    segment is in SLOW_EDGES, in either order; otherwise 1."""
    for (end1, end2, cost) in SLOW_EDGES:
        if (a == end1 and b == end2) or (a == end2 and b == end1):
            return cost
    return 1

def path_cost(position, path):
    """Total cost of driving path from position: the weight
    of every segment, starting with position to path[0]."""
    total = 0
    here = position
    for step in path:
        total = total + weight(here, step)
        here = step
    return total

def check_cost(test_name, position, destination, path, expected_cost):
    """Print PASS if path costs expected_cost, else FAIL with
    the cost and the path; return True on a pass."""
    cost = path_cost(position, path)
    if cost == expected_cost:
        print("  PASS ", test_name, "", path, " cost:", cost)
        return True
    print("  FAIL ", test_name, "", path, " cost:", cost)
    print("          - the cheapest route costs", expected_cost)
    return False

# ===== DIJKSTRA PLANNER (class version) =====
# One Dijkstra object holds the grid size, ITS OWN blocked
# list and ITS OWN slow list, and plans by cost around both:
# the BFS class with the slow-list methods added and the two
# Lesson 6-3 changes inside compute_path(). Same names as the
# function version, with planner. in front; the 5-6 class
# harness tests all three planners without naming one.

def closest(intersections, distance):
    """Return the entry of intersections with the smallest
    value in distance: a plain find-the-minimum loop."""
    best = intersections[0]
    for node in intersections:
        if distance[node] < distance[best]:
            best = node
    return best

class Dijkstra:
    def __init__(self, rows=4, cols=4):
        self.rows = rows
        self.cols = cols
        self.blocked = []
        self.slow_edges = []       # NEW: lives inside the planner

    def add_blocked(self, intersection):
        if intersection not in self.blocked:
            self.blocked.append(intersection)

    def clear_blocked(self):
        self.blocked = []

    def add_slow_edge(self, a, b, cost):
        self.slow_edges.append((a, b, cost))

    def clear_slow_edges(self):
        self.slow_edges = []

    def weight(self, a, b):
        for (end1, end2, cost) in self.slow_edges:
            if (a == end1 and b == end2) or (a == end2 and b == end1):
                return cost
        return 1

    def compute_path(self, position, destination):
        graph = build_graph(self.rows, self.cols, self.blocked)
        if position not in graph or destination not in graph:
            return []
        previous = {position: None}
        distance = {position: 0}
        to_visit = [position]

        while len(to_visit) > 0:
            current = closest(to_visit, distance)
            to_visit.remove(current)
            if current == destination:
                break
            for neighbor in graph[current]:
                new_distance = distance[current] + self.weight(current, neighbor)
                if neighbor not in distance or new_distance < distance[neighbor]:
                    distance[neighbor] = new_distance
                    previous[neighbor] = current
                    if neighbor not in to_visit:
                        to_visit.append(neighbor)

        if destination not in previous:
            return []
        path = []
        current = destination
        while current != position:
            path.append(current)
            current = previous[current]
        path.reverse()
        return path

planner = Dijkstra()

def dijkstra_distances(position):
    """Cheapest cost to every reachable intersection, as a
    dictionary -- the search with no break, like bfs_rings()."""
    graph = build_graph(ROWS, COLS, planner.blocked)
    if position not in graph:
        return {}
    distance = {position: 0}
    to_visit = [position]
    while len(to_visit) > 0:
        current = closest(to_visit, distance)
        to_visit.remove(current)
        for neighbor in graph[current]:
            new_distance = distance[current] + planner.weight(current, neighbor)
            if neighbor not in distance or new_distance < distance[neighbor]:
                distance[neighbor] = new_distance
                if neighbor not in to_visit:
                    to_visit.append(neighbor)
    return distance

def compare(position, destination):
    """Print BFS's and Dijkstra's answers for one trip:
    steps, cost and path, one line each."""

              " cost:", path_cost(position, path), " ", path)
